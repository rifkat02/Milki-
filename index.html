Извлечение пользователя MAX ID только через официальный SDK function extractRealMaxUser() { try { let user = null; const maxObj = window.MAX; if (maxObj) { // Чтение через официально поддерживаемый WebApp SDK if (maxObj.WebApp && maxObj.WebApp.initDataUnsafe && maxObj.WebApp.initDataUnsafe.user) { user = maxObj.WebApp.initDataUnsafe.user; } else if (maxObj.initDataUnsafe && maxObj.initDataUnsafe.user) { user = maxObj.initDataUnsafe.user; } } return user; } catch (e) { console.error('Ошибка получения MAX ID:', e); return null; } } // 2. Обработка нажатия кнопки "Войти с MAX ID" function handleMaxProfileLogin() { const user = extractRealMaxUser(); if (user && (user.id || user.first_name || user.name)) { let savedData = JSON.parse(localStorage.getItem('max_real_user') || '{}'); // Формируем данные профиля const userData = { id: String(user.id || ''), name: `${user.first_name || user.name || ''} ${user.last_name || ''}`.trim() || 'Пользователь MAX', phone: user.phone || savedData.phone || '' }; // Проверка на заблокированных пользователей let blockedUsers = JSON.parse(localStorage.getItem('max_blocked_users')) || []; if (blockedUsers.includes(userData.id)) { alert('Ваш аккаунт заблокирован администратором.'); return; } loginUserToUI(userData); alert(`Успешный вход! Вы вошли как ${userData.name} (ID: ${userData.id})`); } else { alert("Не удалось получить данные MAX ID. Запустите мини-приложение внутри мессенджера MAX."); } } // 3. Функция выполнения HTTPS-запросов к API MAX (согласно обновленной документации) async function maxApiRequest(endpoint, options = {}) { // Получаем access_token или сырую строку initData const accessToken = localStorage.getItem('max_bot_access_token') || ''; const initData = window.MAX?.WebApp?.initData || window.MAX?.initData || ''; // Передача через заголовок Authorization вместо query-параметров const headers = { 'Content-Type': 'application/json', 'Authorization': accessToken ? accessToken : initData, ...(options.headers || {}) }; try { const response = await fetch(`https://api.max.ru/v1/${endpoint}`, { ...options, headers }); return await response.json(); } catch (e) { console.warn('Ошибка вызова API MAX:', e); return null; } }
